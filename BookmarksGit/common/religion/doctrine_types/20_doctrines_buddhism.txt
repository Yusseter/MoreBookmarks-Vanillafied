# Buddhist Canon Doctrines

rice_mahayana_canon_doctrine = {		# Chinese Canon/Dazangjing
	doctrine_group_type = rice_doctrine_buddhist_canon
	index = 0
	divergence = 50
	
	icon = "RICE_mahayana_canon_doctrine"
	
	piety_cost = {
		value = faith_doctrine_cost_mid
		if = {
			limit = { rite_has_doctrine = rice_mahayana_canon_doctrine }
			multiply = 0
		}
		else_if = {
			limit = { rite_has_doctrine = rice_theravada_canon_doctrine }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = rice_vajrayana_canon_doctrine }
			multiply = faith_unchanged_doctrine_cost_mult
		}
		else_if = {
			limit = { rite_has_doctrine = rice_pudgalavada_canon_doctrine }
			multiply = faith_changed_doctrine_cost_mult_three_step
		}
	}
	special_parameters = {
		hostility_level_override = {
			rice_mahayana_canon_doctrine = righteous
		}
	}
	
	parameters = {
		rice_life_liberation_active = yes
		rice_pagodas_unlocked_active = yes
		rice_mahayana_deities = yes
	}

}
